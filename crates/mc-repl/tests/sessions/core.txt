# Rule definitions: a recursive function by rules on literal and typed patterns
mc> rule fib(0) = 0
null
mc> rule fib(1) = 1
null
mc> rule fib(k_: Int) := fib(k - 1) + fib(k - 2)
null
mc> fib(15)
610

mc> fn sgn(x: Int) when x > 0 = 1

# map, and sequence patterns over lists

# Balls print only their correct digits; machine reals win by contagion
mc> 1~30/3
0.33333333333333333333333333333
mc> 2~40^(1/2)
1.414213562373095048801688724209698078570
mc> 1.25~20 + 1

mc> 1~20 + 1.5

# Complex numbers
mc> (1 + 2im)*(3 - im)

# Comparisons are three-valued: undecided stays unevaluated
mc> [1 < 2, 1/3 > 0.3, x == x, x == y, 2 != 2]
[true, true, true, x == y, false]
mc> 1 < 2 and not(2 < 1)
true

# Failures are ordinary results and propagate
mc> 1/0

mc> x + 1/0

# Several statements in one input: the value of the last
mc> a = 2; a^10
